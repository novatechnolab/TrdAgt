// ── Signal Engine ─────────────────────────────────────────────────────────
function computeSignal(row) {
  const { priceChange, oiChange, volRatio } = row;
  const { priceThresh, oiThresh, volMultiplier, signalLogic, confLogic } = state.settings;

  let signal = 'NEUTRAL';
  if (signalLogic === 'price_oi') {
    if (priceChange >= priceThresh && oiChange > 0) signal = 'CALL';
    else if (priceChange <= -priceThresh && oiChange > 0) signal = 'PUT';
    else if (priceChange >= priceThresh) signal = 'CALL';
    else if (priceChange <= -priceThresh) signal = 'PUT';
  } else if (signalLogic === 'price_only') {
    if (priceChange >= priceThresh) signal = 'CALL';
    else if (priceChange <= -priceThresh) signal = 'PUT';
  } else {
    if (oiChange >= oiThresh) signal = priceChange >= 0 ? 'CALL' : 'PUT';
  }

  // Confidence (1–5)
  let conf = 0;
  if (confLogic === 'combined') {
    if (Math.abs(oiChange) >= oiThresh) conf++;
    if (Math.abs(oiChange) >= oiThresh * 1.5) conf++;
    if (Math.abs(priceChange) >= priceThresh) conf++;
    if (Math.abs(priceChange) >= priceThresh * 1.5) conf++;
    if (volRatio >= volMultiplier) conf++;
  } else if (confLogic === 'oi_price') {
    conf = Math.min(5, Math.floor(Math.abs(oiChange) / oiThresh * 2) + (Math.abs(priceChange) >= priceThresh ? 2 : 0));
  } else {
    conf = Math.min(5, Math.floor(Math.abs(oiChange) / oiThresh * 2.5));
  }
  conf = Math.max(1, Math.min(5, conf));

  // Suggest strike (ATM ± 1 strike)
  const atmStrike = Math.round(row.ltp / 50) * 50;
  const strikeOffset = signal === 'CALL' ? atmStrike + 50 : signal === 'PUT' ? atmStrike - 50 : atmStrike;
  const expiry = getNextExpiry();

  return { signal, confidence: conf, suggestedStrike: strikeOffset, expiry };
}

function getNextExpiry() {
  const now = new Date();
  const day = now.getDay();
  const daysToThurs = (4 - day + 7) % 7 || 7;
  const expiry = new Date(now);
  expiry.setDate(now.getDate() + daysToThurs);
  return expiry.toLocaleDateString('en-IN', {day:'2-digit', month:'short'});
}

function generateDemoData() {
  showSpinner(true);
  const now = new Date();
  const timeStr = now.toLocaleTimeString('en-IN', {hour:'2-digit', minute:'2-digit', second:'2-digit', hour12:false});

  // Pick 3–6 random stocks to trigger alerts
  const triggered = [];
  const shuffled = [...FO_STOCKS].sort(() => Math.random() - 0.5).slice(0, Math.floor(Math.random() * 4) + 3);

  shuffled.forEach(stock => {
    const isBull = Math.random() > 0.45;
    const basePrice = basePrices[stock.sym];
    const priceChange = isBull
      ? (Math.random() * 12 + 5) * (Math.random() > 0.3 ? 1 : 0.5)
      : -(Math.random() * 12 + 5) * (Math.random() > 0.3 ? 1 : 0.5);
    const ltp = +(basePrice * (1 + priceChange / 100)).toFixed(2);
    const oiChange = +(Math.random() * 40 + 8).toFixed(1);
    const volRatio = +(Math.random() * 3 + 0.8).toFixed(2);
    const oi = Math.floor(Math.random() * 5000 + 500);

    const row = { sym: stock.sym, sec: stock.sec, ltp, priceChange: +priceChange.toFixed(2), oiChange, volRatio, oi, time: timeStr };
    const sig = computeSignal(row);
    row.signal = sig.signal;
    row.confidence = sig.confidence;
    row.suggestedStrike = sig.suggestedStrike;
    row.expiry = sig.expiry;

    // Check thresholds
    const oiOk = oiChange >= state.settings.oiThresh;
    const priceOk = Math.abs(priceChange) >= state.settings.priceThresh;
    const oiOkMinOI = oi >= state.settings.minOI;

    if ((oiOk || priceOk) && oiOkMinOI) {
      triggered.push(row);

      // Add to alert log
      addAlert(timeStr, stock.sym, priceChange, oiChange, sig.signal);
    }
  });

  // Merge with existing rows (update or add)
  triggered.forEach(r => {
    const idx = state.rows.findIndex(x => x.sym === r.sym);
    if (idx >= 0) state.rows[idx] = r;
    else state.rows.unshift(r);
  });

  renderTable();
  updateStats();
  document.getElementById('lastRefresh').textContent = 'Refreshed ' + timeStr;
  showSpinner(false);
}

// ── Render Table ──────────────────────────────────────────────────────────
function renderTable() {

  let filtered = state.rows.filter(r => {
    if (search && !r.sym.includes(search)) return false;
    if (state.filter === 'CALL' && r.signal !== 'CALL') return false;
    if (state.filter === 'PUT' && r.signal !== 'PUT') return false;
    return true;
  });

  if (filtered.length === 0) {
    body.innerHTML = `<tr><td colspan="10"><div class="empty-state"><div class="empty-icon">🔍</div><div class="empty-msg">No signals match current filters</div></div></td></tr>`;
    document.getElementById('rowCount').textContent = '0 stocks';
    return;
  }

  document.getElementById('rowCount').textContent = filtered.length + ' stocks';

  body.innerHTML = filtered.map(r => {
    const up = r.priceChange >= 0;
    const changeClass = up ? 'up' : 'down';
    const arrow = up ? '▲' : '▼';
    const oiBarPct = Math.min(100, (r.oiChange / 60) * 100);

    if (r.volRatio >= state.settings.volMultiplier) tags.push(`<span class="tag vol-tag">VOL ${r.volRatio}x</span>`);

    // Confidence dots
    const dots = [1,2,3,4,5].map(i => {

    const signalIcon = r.signal === 'CALL' ? 'C' : r.signal === 'PUT' ? 'P' : '–';
    const iconClass = r.signal === 'CALL' ? 'bull-icon' : r.signal === 'PUT' ? 'bear-icon' : 'bear-icon';

    return `<tr class="row-new">
      <td>
        <div class="symbol-cell">
          <div class="symbol-icon ${iconClass}">${signalIcon}</div>

          <span class="oi-val" style="color:${up ? 'var(--bull)' : 'var(--bear)'}">+${r.oiChange}%</span>
        </div>
      </td>
      <td class="right">${volDisp}</td>
      <td><div class="trigger-tags">${tags.join('')}</div></td>
      <td>
        <div class="signal-wrap">
          <span class="signal-badge ${r.signal.toLowerCase()}">${r.signal === 'CALL' ? '▲ CALL' : r.signal === 'PUT' ? '▼ PUT' : '– NEUTRAL'}</span>
        </div>
      </td>
      <td>
        <div class="conf-wrap">
          <div class="conf-dots">${dots}</div>
          <span class="conf-label">${r.confidence}/5</span>
        </div>
      </td>
      <td class="right" style="font-size:11px">
        <div style="color:var(--text)">₹${r.suggestedStrike}</div>
        <div style="color:var(--muted);font-size:9px">${r.expiry}</div>
      </td>
      <td class="right" style="color:var(--muted);font-size:10px">${r.time}</td>
    </tr>`;
  }).join('');
}

// ── Stats ─────────────────────────────────────────────────────────────────
function updateStats() {
  const rows = state.rows;
  const calls = rows.filter(r => r.signal === 'CALL').length;
  const puts = rows.filter(r => r.signal === 'PUT').length;
  const avgOI = rows.length ? (rows.reduce((a,r) => a + r.oiChange, 0) / rows.length).toFixed(1) : 0;
  const high = rows.filter(r => r.confidence >= 4).length;

  document.getElementById('statTotal').textContent = rows.length;
  document.getElementById('statCall').textContent = calls;
  document.getElementById('statPut').textContent = puts;
  document.getElementById('statAvgOI').textContent = avgOI + '%';
  document.getElementById('statHigh').textContent = high;
}

// ── Kite Connect Live ─────────────────────────────────────────────────────
function connectKite() {
  const key = document.getElementById('apiKey').value.trim();
  const token = document.getElementById('accessToken').value.trim();
  if (!key || !token) { alert('Please enter API Key and Access Token'); return; }
  state.apiKey = key;
  state.accessToken = token;
  fetchKiteData();
}

async function fetchKiteData() {
  showSpinner(true);
  document.getElementById('footerMsg').textContent = 'Fetching live data from Kite...';

  try {
    // NOTE: Kite API requires a backend proxy due to CORS.
    // Point this to your Node.js/Python proxy endpoint.
    const PROXY = 'http://localhost:5000';

    // Fetch OI data for F&O instruments
    const instruments = FO_STOCKS.map(s => `NSE:${s.sym}`).join(',');
    const resp = await fetch(`${PROXY}/kite/quote?symbols=${instruments}`, {
      headers: { 'Authorization': `token ${state.apiKey}:${state.accessToken}` }
    });

    if (!resp.ok) throw new Error('API error: ' + resp.status);
    const data = await resp.json();

    processKiteData(data);
    setStatus('connected');
    document.getElementById('connStatus').className = 'connection-status connected';
    document.getElementById('connLabel').textContent = 'Live';
    state.mode = 'live';

    // Schedule refresh
    const interval = parseInt(document.getElementById('refreshInterval').value) * 1000;
    if (state.refreshTimer) clearInterval(state.refreshTimer);
    state.refreshTimer = setInterval(fetchKiteData, interval);

  } catch (e) {
    document.getElementById('footerMsg').textContent = '⚠ Connection failed — ' + e.message + ' (Need backend proxy on localhost:5000)';
    document.getElementById('connStatus').className = 'connection-status disconnected';
    document.getElementById('connLabel').textContent = 'Error';
  }
  showSpinner(false);
}

function processKiteData(kiteResponse) {
  const now = new Date();
  const timeStr = now.toLocaleTimeString('en-IN', {hour12:false});
  const newRows = [];

  Object.entries(kiteResponse.data || {}).forEach(([symbol, quote]) => {
    const sym = symbol.replace('NSE:', '');
    const stock = FO_STOCKS.find(s => s.sym === sym);
    if (!stock) return;

    const ltp = quote.last_price;
    const closeYday = quote.ohlc?.close || ltp;
    const priceChange = +((ltp - closeYday) / closeYday * 100).toFixed(2);
    const oi = quote.oi || 0;
    const oiDayAgo = quote.oi_day_high || oi;
    const oiChange = oiDayAgo > 0 ? +((oi - oiDayAgo) / oiDayAgo * 100).toFixed(1) : 0;
    const vol = quote.volume || 0;
    const avgVol = quote.average_price || 1;
    const volRatio = +(vol / (avgVol * 100000 || 1)).toFixed(2);

    const row = { sym, sec: stock.sec, ltp, priceChange, oiChange, volRatio, oi: oi/100, time: timeStr };
    const sig = computeSignal(row);
    row.signal = sig.signal;
    row.confidence = sig.confidence;
    row.suggestedStrike = sig.suggestedStrike;
    row.expiry = sig.expiry;

    const oiOk = Math.abs(oiChange) >= state.settings.oiThresh;
    const priceOk = Math.abs(priceChange) >= state.settings.priceThresh;

    if (oiOk || priceOk) {
      newRows.push(row);
      addAlert(timeStr, sym, priceChange, oiChange, sig.signal);
    }
  });

  state.rows = newRows.sort((a, b) => Math.abs(b.oiChange) - Math.abs(a.oiChange));
  renderTable();
  updateStats();
  document.getElementById('lastRefresh').textContent = 'Live · ' + timeStr;
}

// ── Export CSV ────────────────────────────────────────────────────────────
function exportCSV() {
  if (state.rows.length === 0) { alert('No data to export'); return; }
  const headers = ['Symbol','Sector','LTP','Change%','OI Change%','Volume Ratio','Signal','Confidence','Strike','Expiry','Time'];
  const rows = state.rows.map(r => [r.sym,r.sec,r.ltp,r.priceChange,r.oiChange,r.volRatio,r.signal,r.confidence,r.suggestedStrike,r.expiry,r.time]);
  const csv = [headers, ...rows].map(r => r.join(',')).join('\n');
  const blob = new Blob([csv], {type:'text/csv'});
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = `oi_spurt_${new Date().toISOString().slice(0,10)}.csv`;
  a.click();
}
