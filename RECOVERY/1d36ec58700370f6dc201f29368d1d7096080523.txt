// ── Connect Live — tries TradeSignal backend first, then standalone ───────────
async function connectLive() {
  showSpinner(true);
  setMsg('Probing backend...');
  hideAuthPanel();

  try {
    // 1) Try TradeSignal integrated backend (/api/oi/health)
    let health = null;
    try {
      const r = await fetch(`${PROXY}/api/oi/health`, { signal: AbortSignal.timeout(3000) });
      if (r.ok) health = await r.json();
    } catch (_) {}

    if (health && health.kite_connected) {
      state.backendType = 'tradesignal';
      setMsg('TradeSignal session OK — running scanner...');
      await runScanner();
      onConnected();
      return;
    }

    // 2) Try standalone OI Spurt server (/health)
    let h2 = null;
    try {
      const r2 = await fetch(`${PROXY}/health`, { signal: AbortSignal.timeout(3000) });
      if (r2.ok) h2 = await r2.json();
    } catch (_) {}

    if (h2 && h2.connected) {
      state.backendType = 'standalone';
      setMsg('Standalone OI server connected — running scanner...');
      await runScanner();
      onConnected();
      return;
    }

    // 3) Not connected — show auth panel
    if (h2 && !h2.connected) {
      setMsg('⚠ Kite not authenticated — enter API Key + Access Token below.');
      showAuthPanel();
      setConnBadge('disconnected');
    } else {
      setMsg('⚠ No backend found at localhost:5000 — start TradeSignal or OI Spurt server.');
      setConnBadge('disconnected');
    }
  } catch (e) {
    setMsg('⚠ Error: ' + e.message);
    setConnBadge('disconnected');
  }
  showSpinner(false);
}

// ── Authenticate and connect (manual key entry) ───────────────────────────────
async function authAndConnect() {
  const apiKey = document.getElementById('authApiKey').value.trim();
  const token  = document.getElementById('authToken').value.trim();
  if (!apiKey || !token) { setMsg('⚠ Enter both API Key and Access Token.'); return; }
  showSpinner(true);
  setMsg('Authenticating...');
  try {
    const r = await fetch(`${PROXY}/kite/auth`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ apiKey, accessToken: token })
    });
    const d = await r.json();
    if (!r.ok || d.error) throw new Error(d.error || 'Auth failed');
    hideAuthPanel();
    state.backendType = 'standalone';
    setMsg('Authenticated — running scanner...');
    await runScanner();
    onConnected();
  } catch (e) {
    setMsg('⚠ Auth failed: ' + e.message);
    setConnBadge('disconnected');
  }
  showSpinner(false);
}

// ── Main scanner fetch — calls /kite/scanner with all params ─────────────────
async function runScanner() {
  showSpinner(true);
  const s = state.settings;
  // TradeSignal uses /kite/scanner (same endpoint structure, registered in server.py)
  const url = `${PROXY}/api/oi/scanner?oiThresh=${s.oiThresh}&priceThresh=${s.priceThresh}&minOI=100&volMult=${s.volMult}&signalLogic=${s.signalLogic}`;
  try {
    const r = await fetch(url);
    if (!r.ok) throw new Error('HTTP ' + r.status);
    const data = await r.json();
    if (data.error) throw new Error(data.error);
    processScannerData(data);
    document.getElementById('lastRefresh').textContent =
      'Live · ' + new Date().toLocaleTimeString('en-IN', { hour12: false }) +
      ` · ${data.count} signals`;
    setMsg(`▶ Live — ${data.count} signals · ${data.call_count} CALL · ${data.put_count} PUT`);
    setConnBadge('connected');
  } catch (e) {
    setMsg('⚠ Scanner error: ' + e.message);
    setConnBadge('disconnected');
    throw e;
  } finally {
    showSpinner(false);
  }
}

function generateDemoData() {
  const timeStr = new Date().toLocaleTimeString('en-IN', { hour12: false });
  const s = state.settings;
  const shuffled = [...DEMO_STOCKS].sort(() => Math.random() - 0.5).slice(0, 20);
  const rows = [];

  shuffled.forEach(stock => {
    basePrices[stock.sym] = basePrices[stock.sym] || (Math.random() * 2000 + 200);
    const drift = (Math.random() - 0.48) * 0.018;
    basePrices[stock.sym] = +(basePrices[stock.sym] * (1 + drift)).toFixed(2);
    const ltp         = basePrices[stock.sym];
    const priceChange = +((Math.random() - 0.45) * 20).toFixed(2);
    const oiChange    = +((Math.random() - 0.38) * 45).toFixed(1);
    const volRatio    = +(Math.random() * 3.5 + 0.5).toFixed(2);

    const oiOk    = Math.abs(oiChange) >= s.oiThresh;
    const priceOk = Math.abs(priceChange) >= s.priceThresh;
    if (!oiOk && !priceOk) return;

    // Mirror server-side signal logic
    let signal = 'NEUTRAL';
    if (s.signalLogic === 'price_oi') {
      if (priceChange >= s.priceThresh && oiChange > 0)       signal = 'CALL';
      else if (priceChange <= -s.priceThresh && oiChange > 0) signal = 'PUT';
      else if (priceChange >= s.priceThresh)                  signal = 'CALL';
      else if (priceChange <= -s.priceThresh)                 signal = 'PUT';
    } else if (s.signalLogic === 'price_only') {
      signal = priceChange >= s.priceThresh ? 'CALL' : priceChange <= -s.priceThresh ? 'PUT' : 'NEUTRAL';
    } else {
      signal = oiChange >= s.oiThresh ? 'CALL' : oiChange <= -s.oiThresh ? 'PUT' : 'NEUTRAL';
    }

    // Mirror server-side confidence
    let conf = 0;
    if (Math.abs(oiChange) >= s.oiThresh)        conf++;
    if (Math.abs(oiChange) >= s.oiThresh * 1.5)  conf++;
    if (Math.abs(priceChange) >= s.priceThresh)  conf++;
    if (Math.abs(priceChange) >= s.priceThresh * 1.5) conf++;
    if (volRatio >= s.volMult)                   conf++;
    conf = Math.max(1, Math.min(5, conf));

    const strike = Math.round(ltp / 50) * 50;
    rows.push({
      sym: stock.sym, sec: stock.sec, ltp, priceChange, oiChange,
      oiLots: Math.floor(Math.random() * 50000),
      oiBaseline: Math.floor(Math.random() * 40000),
      volRatio, signal, confidence: conf,
      suggestedStrike: signal === 'CALL' ? strike + 50 : strike - 50,
      expiry: getNextExpiry(),
      triggeredByOI: oiOk, triggeredByPx: priceOk,
      time: timeStr
    });
  });

  state.rows = rows.sort((a, b) => Math.abs(b.oiChange) - Math.abs(a.oiChange));
  if (rows.length) addAlert(timeStr, rows[0].sym, rows[0].priceChange, rows[0].oiChange, rows[0].signal);
  renderTable();
  updateStats({ call_count: rows.filter(r=>r.signal==='CALL').length, put_count: rows.filter(r=>r.signal==='PUT').length });
  document.getElementById('lastRefresh').textContent = 'Demo · ' + timeStr + ` · ${rows.length} signals`;
}

function getNextExpiry() {
  const d = new Date();
  d.setDate(d.getDate() + ((4 - d.getDay() + 7) % 7 || 7));
  return `${d.getDate().toString().padStart(2,'0')}-${['JAN','FEB','MAR','APR','MAY','JUN','JUL','AUG','SEP','OCT','NOV','DEC'][d.getMonth()]}`;
}

// ── Alerts ────────────────────────────────────────────────────────────────────
let alertCnt = 0;
const seenAlerts = new Set();
function addAlert(time, sym, px, oi, signal) {
  const key = `${sym}-${Math.round(oi)}-${Math.round(px)}`;
  if (seenAlerts.has(key)) return;
  seenAlerts.add(key);
  if (seenAlerts.size > 200) seenAlerts.clear();

  const log = document.getElementById('alertLog');
  const empty = log.querySelector('.alert-empty');
  if (empty) empty.remove();
  const cls = signal === 'CALL' ? 'var(--bull)' : signal === 'PUT' ? 'var(--bear)' : 'var(--muted)';
  const el = document.createElement('div');
  el.className = 'alert-item';
  el.innerHTML = `<span class="alert-time">${time}</span>` +
    `<span class="alert-sym" style="color:${cls}">${sym}</span>` +
    `<span class="alert-msg">Px ${px>=0?'+':''}${px}% · OI Spurt ${oi>=0?'+':''}${oi}% · <b>${signal}</b></span>`;
  log.insertBefore(el, log.firstChild);
  while (log.children.length > 50) log.removeChild(log.lastChild);
  alertCnt++;
  document.getElementById('alertCount').textContent = alertCnt + ' alerts';
}

// ── Render Table ──────────────────────────────────────────────────────────────
function renderTable() {
  const body   = document.getElementById('tableBody');
  const search = document.getElementById('searchBox').value.toLowerCase();
  const fOI    = parseFloat(document.getElementById('fOI').value) || 0;
  const fPx    = parseFloat(document.getElementById('fPx').value) || 0;
  const sortBy = document.getElementById('sortBy').value;

  let rows = state.rows.filter(r => {
    if (search && !r.sym.toLowerCase().includes(search)) return false;
    if (state.filter !== 'ALL' && r.signal !== state.filter) return false;
    if (Math.abs(r.oiChange) < fOI)   return false;
    if (Math.abs(r.priceChange) < fPx) return false;
    if (r.confidence < state.minConf) return false;
    return true;
  });

  if (sortBy === 'px')   rows.sort((a,b) => Math.abs(b.priceChange) - Math.abs(a.priceChange));
  else if (sortBy === 'conf') rows.sort((a,b) => b.confidence - a.confidence);
  else rows.sort((a,b) => Math.abs(b.oiChange) - Math.abs(a.oiChange));

  document.getElementById('rowCount').textContent = rows.length + ' stocks';

  if (!rows.length) {
    const msg = state.rows.length ? 'No signals match current filters' : 'Click CONNECT LIVE or DEMO MODE to begin';
    body.innerHTML = `<tr><td colspan="10"><div class="empty-state"><div class="empty-icon">${state.rows.length?'🔍':'📡'}</div><div class="empty-msg">${msg}</div></div></td></tr>`;
    return;
  }

  const s = state.settings;
  body.innerHTML = rows.map(r => {
    const iconCls  = r.signal==='CALL' ? 'bull-icon' : r.signal==='PUT' ? 'bear-icon' : '';
    const sigIcon  = r.signal==='CALL' ? '▲' : r.signal==='PUT' ? '▼' : '◆';
    const pxCls    = r.priceChange >= 0 ? 'up' : 'dn';
    const barPct   = Math.min(100, Math.abs(r.oiChange) * 2);
    const barCls   = r.oiChange >= 0 ? 'bull' : 'bear';
    const oiColor  = r.oiChange >= 0 ? 'var(--bull)' : 'var(--bear)';
    const sigCls   = r.signal==='CALL' ? 'call' : r.signal==='PUT' ? 'put' : 'neutral';
    const dots     = Array.from({length:5}, (_,i) =>
      `<div class="conf-dot ${i<r.confidence?(r.signal==='CALL'?'fb':r.signal==='PUT'?'fd':'fw'):''}"></div>`
    ).join('');
    // Tags from backend triggered_by flags
    const tags = [
      r.triggeredByOI  ? `<span class="tag oi-tag">OI ${r.oiChange>=0?'+':''}${r.oiChange.toFixed(0)}%</span>` : '',
      r.triggeredByPx  ? `<span class="tag px-tag">PX ${r.priceChange>=0?'+':''}${r.priceChange}%</span>` : '',
      r.volRatio>=s.volMult ? `<span class="tag vol-tag">VOL ${r.volRatio}x</span>` : ''
    ].filter(Boolean).join('');
    const oiLots = r.oiLots ? Number(r.oiLots).toLocaleString('en-IN') : '—';
    return `<tr class="row-new">
      <td><div class="sym-cell"><div class="sym-icon ${iconCls}">${sigIcon}</div><div><div class="sym-name">${r.sym}</div><div class="sym-sec">${r.sec}</div></div></div></td>
      <td class="r">${r.ltp ? r.ltp.toFixed(2) : '—'}</td>
      <td class="r"><span class="chg-badge ${pxCls}">${r.priceChange>=0?'+':''}${r.priceChange}%</span></td>
      <td class="r" style="color:${oiColor};font-weight:600">${r.oiChange>=0?'+':''}${r.oiChange}%</td>
      <td><div class="oi-bar-wrap"><div class="oi-bar-bg"><div class="oi-bar-fill ${barCls}" style="width:${barPct}%"></div></div><span class="oi-val" style="color:${oiColor}">${Math.abs(r.oiChange).toFixed(0)}%</span></div></td>
      <td class="r">${r.volRatio}x</td>
      <td><span class="sig-badge ${sigCls}">${sigIcon} ${r.signal}</span></td>
      <td><div class="conf-dots">${dots}</div></td>
      <td><div class="trigger-tags">${tags||'<span class="tag">—</span>'}</div></td>
      <td class="r" style="color:var(--muted)">${r.time}</td>
    </tr>`;
  }).join('');
}

// ── Stats ─────────────────────────────────────────────────────────────────────
function updateStats(data) {
  const rows  = state.rows;
  const calls = data?.call_count ?? rows.filter(r=>r.signal==='CALL').length;
  const puts  = data?.put_count  ?? rows.filter(r=>r.signal==='PUT').length;
  const avgOI = rows.length ? (rows.reduce((s,r)=>s+Math.abs(r.oiChange),0)/rows.length).toFixed(1) : 0;
  const top   = rows.length ? rows[0].sym : '—';
  document.getElementById('statTotal').textContent = rows.length;
  document.getElementById('statCall').textContent  = calls;
  document.getElementById('statPut').textContent   = puts;
  document.getElementById('statAvgOI').textContent = avgOI + '%';
  document.getElementById('statTop').textContent   = top;
}
