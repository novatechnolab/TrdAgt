function isMarketOpen() {
  const now = new Date();
  const ist = new Date(now.toLocaleString('en-US', {timeZone:'Asia/Kolkata'}));
  const day = ist.getDay(); // 0=Sun,6=Sat
  if (day === 0 || day === 6) return false;
  const h = ist.getHours(), m = ist.getMinutes();
  const mins = h * 60 + m;
  return mins >= 9*60+15 && mins <= 15*60+30;
}

function isNSEHoliday(date) {
  // date is a local Date object already in IST context
  const key = `${date.getFullYear()}-${String(date.getMonth()+1).padStart(2,'0')}-${String(date.getDate()).padStart(2,'0')}`;
  return NSE_HOLIDAYS.has(key);
}

function getSessionTradingDay() {
  // Returns the IST date we should display data for.
  // Skips weekends AND NSE market holidays.
  // state.sessionDayOffset allows runtime walk-back when data is empty.
  const now = new Date();
  const ist = new Date(now.toLocaleString('en-US', {timeZone:'Asia/Kolkata'}));
  const dow = ist.getDay(); // 0=Sun, 6=Sat
  const mins = ist.getHours() * 60 + ist.getMinutes();
  const marketStart = 9 * 60 + 15; // 09:15 IST

  // Today is a valid live-data day: weekday, past market open, not a holiday
  const todayIsValid = (dow !== 0 && dow !== 6 && mins >= marketStart && !isNSEHoliday(ist));

  // tradingDaysBack = how many valid trading days to walk back
  let tradingDaysBack = (state?.sessionDayOffset || 0);
  if (!todayIsValid) tradingDaysBack++; // today itself is not valid, need at least 1 step

  if (tradingDaysBack === 0) return new Date(ist); // today is valid, no offset

  // Walk back until we've found 'tradingDaysBack' valid trading days
  let dayToCheck = new Date(ist);
  let found = 0;
  for (let i = 0; i < 15; i++) {
    dayToCheck.setDate(dayToCheck.getDate() - 1);
    const d = dayToCheck.getDay();
    if (d !== 0 && d !== 6 && !isNSEHoliday(dayToCheck)) {
      found++;
      if (found >= tradingDaysBack) return new Date(dayToCheck);
    }
  }
  return dayToCheck;
}


function getSessionDateRange(symbol, tf) {
  const sessionDate = getSessionTradingDay();

  // Set from = 9:15 IST, to = 15:30 IST (full session)
  const from = new Date(sessionDate);
  from.setHours(9, 15, 0, 0);

  const to = new Date(sessionDate);
  to.setHours(15, 30, 0, 0);

  return { from, to, sessionDate };
}

function calcEMA(closes, period) {
  if (closes.length < period) return Array(closes.length).fill(null);
  const k = 2/(period+1);
  let ema = [closes.slice(0,period).reduce((a,b)=>a+b,0)/period];
  for (let i=period;i<closes.length;i++) ema.push(closes[i]*k+ema[ema.length-1]*(1-k));
  return [...Array(closes.length-ema.length).fill(null),...ema];
}

function calcVWAP(candles) {
  let cpv=0,cv=0;
  return candles.map(c=>{const tp=(c.high+c.low+c.close)/3;cpv+=tp*c.volume;cv+=c.volume;return cv?cpv/cv:null;});
}

function calcRSI(closes,period=14) {
  if (closes.length<period+1) return Array(closes.length).fill(null);
  const d=[],g=[],l=[];
  for(let i=1;i<closes.length;i++){const v=closes[i]-closes[i-1];d.push(v);g.push(v>0?v:0);l.push(v<0?-v:0);}
  let ag=g.slice(0,period).reduce((a,b)=>a+b,0)/period;
  let al=l.slice(0,period).reduce((a,b)=>a+b,0)/period;
  const r=[...Array(period+1).fill(null),100-100/(1+ag/(al||.0001))];
  for(let i=period;i<g.length;i++){ag=(ag*(period-1)+g[i])/period;al=(al*(period-1)+l[i])/period;r.push(100-100/(1+ag/(al||.0001)));}
  return r;
}

function calcMACD(closes,fast=12,slow=26,signal=9) {
  const ef=calcEMA(closes,fast),es=calcEMA(closes,slow);
  const ml=closes.map((_,i)=>(ef[i]!=null&&es[i]!=null)?ef[i]-es[i]:null);
  const valid=ml.filter(v=>v!=null);
  if(valid.length<signal) return {macdLine:ml,sigFull:Array(ml.length).fill(null),hist:Array(ml.length).fill(null)};
  const sl=calcEMA(valid,signal);
  const pad=ml.length-sl.length;
  const sf=[...Array(pad).fill(null),...sl];
  const hist=ml.map((v,i)=>(v!=null&&sf[i]!=null)?v-sf[i]:null);
  return {macdLine:ml,sigFull:sf,hist};
}

function detectSwings(candles,lb=3) {
  const H=[],L=[];
  for(let i=lb;i<candles.length-lb;i++){
    const w=candles.slice(i-lb,i+lb+1);
    if(candles[i].high===Math.max(...w.map(c=>c.high))) H.push(i);
    if(candles[i].low===Math.min(...w.map(c=>c.low)))   L.push(i);
  }
  return {H,L};
}

function detectBOS(candles) {
  if(candles.length<10) return 'none';
  const {H,L}=detectSwings(candles);
  const last=candles[candles.length-1].close;
  if(H.length && last>candles[H[H.length-1]].high) return 'bullish';
  if(L.length && last<candles[L[L.length-1]].low)  return 'bearish';
  return 'none';
}

function detectCHoCH(candles) {
  if(candles.length<15) return 'none';
  const {H,L}=detectSwings(candles);
  const last=candles[candles.length-1].close;
  if(H.length>=2 && candles[H[H.length-1]].high<candles[H[H.length-2]].high)
    if(last>candles[H[H.length-2]].high) return 'bullish';
  if(L.length>=2 && candles[L[L.length-1]].low>candles[L[L.length-2]].low)
    if(last<candles[L[L.length-2]].low) return 'bearish';
  return 'none';
}

function detectBOSEvents(candles) {
  // Returns array of {idx, type} for chart markers
  const events=[];
  const {H,L}=detectSwings(candles);
  for(let i=1;i<H.length;i++) {
    const sh=candles[H[i-1]].high;
    // Find first candle after H[i-1] that closes above sh
    for(let j=H[i-1]+1;j<candles.length;j++) {
      if(candles[j].close>sh){events.push({idx:j,type:'bullish',price:candles[j].high});break;}
    }
  }
  for(let i=1;i<L.length;i++) {
    const sl=candles[L[i-1]].low;
    for(let j=L[i-1]+1;j<candles.length;j++) {
      if(candles[j].close<sl){events.push({idx:j,type:'bearish',price:candles[j].low});break;}
    }
  }
  return events.slice(-4);
}

function detectCHoCHEvents(candles) {
  const events=[];
  const {H,L}=detectSwings(candles);
  for(let i=1;i<H.length;i++) {
    if(candles[H[i]].high<candles[H[i-1]].high) {
      for(let j=H[i]+1;j<candles.length;j++) {
        if(candles[j].close>candles[H[i-1]].high){events.push({idx:j,type:'bullish',price:candles[j].high});break;}
      }
    }
  }
  for(let i=1;i<L.length;i++) {
    if(candles[L[i]].low>candles[L[i-1]].low) {
      for(let j=L[i]+1;j<candles.length;j++) {
        if(candles[j].close<candles[L[i-1]].low){events.push({idx:j,type:'bearish',price:candles[j].low});break;}
      }
    }
  }
  return events.slice(-4);
}

function detectOrderBlocks(candles) {
  const obs=[];
  for(let i=2;i<candles.length-1;i++) {
    const c=candles[i],n=candles[i+1];
    if(c.close<c.open&&n.close>c.high) obs.push({type:'bull',top:c.open,bottom:c.low,idx:i,tested:false});
    if(c.close>c.open&&n.close<c.low)  obs.push({type:'bear',top:c.high,bottom:c.close,idx:i,tested:false});
  }
  const last=candles[candles.length-1];
  obs.forEach(ob=>{
    if(ob.type==='bull'&&last.low<=ob.top&&last.low>=ob.bottom) ob.tested=true;
    if(ob.type==='bear'&&last.high>=ob.bottom&&last.high<=ob.top) ob.tested=true;
  });
  return obs.slice(-6);
}

function detectFVGs(candles) {
  const fvgs=[];
  for(let i=2;i<candles.length;i++) {
    const p=candles[i-2],c=candles[i];
    if(c.low>p.high) fvgs.push({type:'bull',top:c.low,bottom:p.high,idx:i});
    if(c.high<p.low) fvgs.push({type:'bear',top:p.low,bottom:c.high,idx:i});
  }
  return fvgs.slice(-5);
}

function isVolSpike(candles,period=20) {
  if(candles.length<period+1) return false;
  const avg=candles.slice(-period-1,-1).reduce((s,c)=>s+c.volume,0)/period;
  return candles[candles.length-1].volume>avg*1.5;
}

function htfBias(candles15) {
  if(!candles15||candles15.length<50) return 'neutral';
  const c=candles15.map(x=>x.close);
  const e21=calcEMA(c,21),e50=calcEMA(c,50);
  const last=c[c.length-1],l21=e21[e21.length-1],l50=e50[e50.length-1];
  if(l21&&l50){
    if(last>l21&&l21>l50) return 'bullish';
    if(last<l21&&l21<l50) return 'bearish';
  }
  return 'neutral';
}

function calcATR(candles,period=14) {
  const trs=candles.slice(1).map((c,i)=>Math.max(c.high-c.low,Math.abs(c.high-candles[i].close),Math.abs(c.low-candles[i].close)));
  if(!trs.length) return 1;
  return trs.slice(-period).reduce((a,b)=>a+b,0)/Math.min(period,trs.length);
}

function scoreSignal(candles, ema21, ema50, vwap, rsi, macd, bos, choch, htf, volSpike) {
  if(candles.length<2) return {score:0,direction:'WAIT',reason:[],bullScore:0,bearScore:0};
  const close=candles[candles.length-1].close;
  let bull=0,bear=0; const reason=[];
  const e21=ema21[ema21.length-1],e50=ema50[ema50.length-1];
  if(e21&&e50){
    if(close>e21&&e21>e50){bull+=2;reason.push('Price > EMA21 > EMA50 ✓');}
    else if(close<e21&&e21<e50){bear+=2;reason.push('Price < EMA21 < EMA50 ✓');}
  }
  const vw=vwap[vwap.length-1];
  if(vw){if(close>vw){bull+=1;reason.push('Above VWAP ✓');}else{bear+=1;reason.push('Below VWAP ✓');}}
  const r=rsi[rsi.length-1];
  if(r){
    if(r>55&&r<75){bull+=1;reason.push(`RSI ${r.toFixed(0)} bullish`);}
    else if(r<45&&r>25){bear+=1;reason.push(`RSI ${r.toFixed(0)} bearish`);}
  }
  const mh=macd.hist[macd.hist.length-1],mhp=macd.hist[macd.hist.length-2];
  if(mh!=null){
    if(mh>0&&mhp<0){bull+=2;reason.push('MACD bullish crossover ✓');}
    else if(mh<0&&mhp>0){bear+=2;reason.push('MACD bearish crossover ✓');}
    else if(mh>0){bull+=1;reason.push('MACD positive');}
    else{bear+=1;reason.push('MACD negative');}
  }
  if(choch==='bullish'){bull+=2;reason.push('CHoCH bullish ✓');}
  else if(choch==='bearish'){bear+=2;reason.push('CHoCH bearish ✓');}
  if(bos==='bullish'){bull+=1;reason.push('BOS bullish ✓');}
  else if(bos==='bearish'){bear+=1;reason.push('BOS bearish ✓');}
  if(htf==='bullish'){bull+=2;reason.push('15m HTF bullish ✓');}
  else if(htf==='bearish'){bear+=2;reason.push('15m HTF bearish ✓');}
  if(volSpike){if(bull>bear){bull+=1;reason.push('Volume spike (bull conf)');}else{bear+=1;reason.push('Volume spike (bear conf)');}}
  let direction='WAIT',score=0;
  if(bull>=5&&bull>bear){direction='BUY';score=Math.round(bull/SCORE_MAX*10);}
  else if(bear>=5&&bear>bull){direction='SELL';score=Math.round(bear/SCORE_MAX*10);}
  return {score,direction,reason,bullScore:bull,bearScore:bear};
}

function calcTPSL(candles,direction,obs) {
  const last=candles[candles.length-1];
  const atr=calcATR(candles,14);
  let tp,sl;
  if(direction==='BUY'){
    sl=last.close-atr*1.5; tp=last.close+atr*2.5;
    const bOB=obs.filter(o=>o.type==='bull'&&o.top<last.close).sort((a,b)=>b.top-a.top)[0];
    if(bOB) sl=Math.max(sl,bOB.bottom);
  } else {
    sl=last.close+atr*1.5; tp=last.close-atr*2.5;
    const bOB=obs.filter(o=>o.type==='bear'&&o.bottom>last.close).sort((a,b)=>a.bottom-b.bottom)[0];
    if(bOB) sl=Math.min(sl,bOB.top);
  }
  return {entry:last.close.toFixed(2),tp:tp.toFixed(2),sl:sl.toFixed(2)};
}

function renderChart(candles, ema21, ema50, vwap, obs, fvgs, chochEvents, bosEvents) {
  console.log(`[CHART] renderChart called with ${candles.length} candles`);
  if(!chart) {
    console.error(`[CHART] chart object is null!`);
    return;
  }
  
  try {
    // Normalize timestamps to numeric milliseconds for luxon adapter
    const timestamps=candles.map(c=>{
      const ts=c.timestamp;
      if(typeof ts==='number') return ts;  // Already milliseconds
      if(typeof ts==='string') return new Date(ts).getTime();  // ISO string
      return ts;  // fallback
    });

    // --- FIX: Lock X-axis to 09:15–15:30 IST for the correct session date ---
    const { sessionDate } = getSessionDateRange(state.symbol, state.tf);
    const xMin = new Date(sessionDate); xMin.setHours(9, 15, 0, 0);
    const xMax = new Date(sessionDate); xMax.setHours(15, 30, 0, 0);
    SESSION_X_MIN = xMin.getTime();
    SESSION_X_MAX = xMax.getTime();
    chart.options.scales.x.min = SESSION_X_MIN;
    chart.options.scales.x.max = SESSION_X_MAX;
    // Update stepSize to match the current timeframe so ticks snap correctly
    chart.options.scales.x.time.stepSize = state.tf;
    // Update closed-banner to show which date is displayed
    const banner = document.getElementById('closedBanner');
    if (banner && !isMarketOpen()) {
      const dateStr = sessionDate.toLocaleDateString('en-IN', {weekday:'short', day:'numeric', month:'short', year:'numeric', timeZone:'Asia/Kolkata'});
      banner.textContent = `⏸ MARKET CLOSED — Showing ${dateStr} session data · NSE opens 09:15 IST Mon–Fri`;
    }
    // -------------------------------------------------------------------------
    
    chart._obData=obs; chart._fvgData=fvgs; chart._candleData=candles;
    chart._chochData=chochEvents; chart._bosData=bosEvents;
    chart._prevClose=state.prevClose||null;

    // Build datasets with valid data
    const datasets=[];
    
    // Candlestick data - core price action with numeric timestamps
    const candleData=candles.map((c,i)=>({
      x:timestamps[i],
      o:parseFloat(c.open),
      h:parseFloat(c.high),
      l:parseFloat(c.low),
      c:parseFloat(c.close)
    }));
    datasets.push({
      label:'Price',
      type:'candlestick',
      data:candleData,
      color:{up:'#00e676',down:'#ff3d57',unchanged:'#4a5a6b'},
      borderColor:{up:'#00e676',down:'#ff3d57',unchanged:'#4a5a6b'}
    });
    
    // EMA21 - filter nulls, use numeric x values
    const ema21Data=ema21.map((v,i)=>v!=null?{x:timestamps[i],y:parseFloat(v)}:null).filter(d=>d!=null);
    if(ema21Data.length) {
      datasets.push({
        label:'EMA21',
        type:'line',
        data:ema21Data,
        borderColor:'#ffc400',
        borderWidth:1.5,
        pointRadius:0,
        tension:.3,
        spanGaps:true
      });
    }
    
    // EMA50 - filter nulls, use numeric x values
    const ema50Data=ema50.map((v,i)=>v!=null?{x:timestamps[i],y:parseFloat(v)}:null).filter(d=>d!=null);
    if(ema50Data.length) {
      datasets.push({
        label:'EMA50',
        type:'line',
        data:ema50Data,
        borderColor:'#60a5fa',
        borderWidth:1.5,
        pointRadius:0,
        tension:.3,
        spanGaps:true
      });
    }
    
    // VWAP - filter nulls, use numeric x values
    const vwapData=vwap.map((v,i)=>v!=null?{x:timestamps[i],y:parseFloat(v)}:null).filter(d=>d!=null);
    if(vwapData.length) {
      datasets.push({
        label:'VWAP',
        type:'line',
        data:vwapData,
        borderColor:'#a78bfa',
        borderWidth:1.5,
        borderDash:[4,3],
        pointRadius:0,
        tension:.3,
        spanGaps:true
      });
    }

    // CHoCH markers
    if(chochEvents.length){
      datasets.push({
        label:'CHoCH',
        type:'scatter',
        data:chochEvents.map(e=>({x:timestamps[e.idx],y:parseFloat(e.price)})),
        backgroundColor:'#ff9100',
        pointRadius:7,
        pointStyle:'triangle'
      });
    }
    
    // BOS markers
    if(bosEvents.length){
      datasets.push({
        label:'BOS',
        type:'scatter',
        data:bosEvents.map(e=>({x:timestamps[e.idx],y:parseFloat(e.price)})),
        backgroundColor:'#c084fc',
        pointRadius:6,
        pointStyle:'rectRot'
      });
    }
    
    console.log(`[CHART] Setting ${datasets.length} datasets, updating chart...`);
    chart.data.datasets=datasets;
    chart.update('none');
    console.log(`[CHART] Update complete`);
  } catch(err) {
    console.error(`[CHART] renderChart error:`, err);
    throw err;
  }
}

const crosshairPlugin = {
  id: 'crosshairPlugin',
  afterDraw(chart) {
    const ch = chart._crosshair;
    if (!ch || !ch.visible) return;
    const {ctx, scales:{x, y}} = chart;
    const xl=x.left, xr=x.right, yt=y.top, yb=y.bottom;
    const {px, py} = ch;
    if (px < xl || px > xr || py < yt || py > yb) return;

    ctx.save();
    ctx.setLineDash([4, 4]);
    ctx.lineWidth = 1;
    ctx.strokeStyle = 'rgba(0,212,255,0.55)';

    // Vertical hair
    ctx.beginPath(); ctx.moveTo(px, yt); ctx.lineTo(px, yb); ctx.stroke();
    // Horizontal hair
    ctx.beginPath(); ctx.moveTo(xl, py); ctx.lineTo(xr, py); ctx.stroke();
    ctx.setLineDash([]);

    // Price tag on right Y-axis margin
    const price = y.getValueForPixel(py);
    if (price != null) {
      const tag = price.toFixed(2);
      const tw = ctx.measureText(tag).width + 10;
      ctx.fillStyle = '#00d4ff';
      ctx.fillRect(xr+1, py-9, tw, 18);
      ctx.fillStyle = '#000';
      ctx.font = 'bold 9px JetBrains Mono';
      ctx.textAlign = 'left';
      ctx.fillText(tag, xr+6, py+4);
    }
    // Time tag on bottom X-axis margin
    const tsVal = x.getValueForPixel(px);
    if (tsVal != null) {
      const timeStr = new Date(tsVal).toLocaleTimeString('en-IN',{timeZone:'Asia/Kolkata',hour:'2-digit',minute:'2-digit',hour12:false});
      const tw2 = ctx.measureText(timeStr).width + 10;
      ctx.fillStyle = '#00d4ff';
      ctx.fillRect(px - tw2/2, yb+1, tw2, 16);
      ctx.fillStyle = '#000';
      ctx.font = 'bold 9px JetBrains Mono';
      ctx.textAlign = 'center';
      ctx.fillText(timeStr, px, yb+12);
    }
    ctx.restore();
  }
};

// ZONE PLUGIN  (fills inside chart, labels in right margin)
// ============================================================
const zonePlugin = {
  id:'zonePlugin',
  afterDraw(chart){
    const obs=chart._obData||[],fvgs=chart._fvgData||[];
    const candles=chart._candleData||[];
    const chochEvts=chart._chochData||[],bosEvts=chart._bosData||[];
    const prevClose=chart._prevClose||null;
    if(!candles.length) return;
    const {ctx,scales:{x,y}}=chart;
    const xl=x.left,xr=x.right,yt=y.top,yb=y.bottom;
    const cw=chart.width;

    const tsToX=ts=>{
      const ms=typeof ts==='number'?ts:(typeof ts==='string'?new Date(ts).getTime():ts);
      return x.getPixelForValue(ms);
    };
    const idxToX=i=>{ return tsToX(candles[i]?.timestamp); };

    // ── PHASE 1: clipped to plot area — fills & borders only ──
    ctx.save();
    ctx.beginPath(); ctx.rect(xl,yt,xr-xl,yb-yt); ctx.clip();

    // Prev close dashed line
    if(prevClose!=null){
      const py=y.getPixelForValue(prevClose);
      if(py>=yt&&py<=yb){
        ctx.strokeStyle='rgba(255,196,0,0.6)';
        ctx.lineWidth=1; ctx.setLineDash([6,4]);
        ctx.beginPath();ctx.moveTo(xl,py);ctx.lineTo(xr,py);ctx.stroke();
        ctx.setLineDash([]);
      }
    }
    // OB fills + solid left edge
    obs.forEach(ob=>{
      const isBull=ob.type==='bull';
      const color=isBull?[0,230,118]:[255,61,87];
      const x1=idxToX(ob.idx); if(x1>xr) return;
      const y1=y.getPixelForValue(ob.top),y2=y.getPixelForValue(ob.bottom);
      const yTop=Math.min(y1,y2),h=Math.abs(y2-y1);
      ctx.fillStyle=`rgba(${color},${ob.tested?0.04:0.10})`;
      ctx.fillRect(x1,yTop,xr-x1,h);
      ctx.strokeStyle=`rgba(${color},0.85)`;ctx.lineWidth=2;ctx.setLineDash([]);
      ctx.beginPath();ctx.moveTo(x1,yTop);ctx.lineTo(x1,yTop+h);ctx.stroke();
      ctx.strokeStyle=`rgba(${color},0.20)`;ctx.lineWidth=1;ctx.setLineDash([2,4]);
      ctx.strokeRect(x1,yTop,xr-x1,h);ctx.setLineDash([]);
    });
    // FVG fills
    fvgs.forEach(fvg=>{
      const isBull=fvg.type==='bull';
      const x1=idxToX(fvg.idx); if(x1>xr) return;
      const y1=y.getPixelForValue(fvg.top),y2=y.getPixelForValue(fvg.bottom);
      const yTop=Math.min(y1,y2),h=Math.abs(y2-y1);
      ctx.fillStyle=isBull?'rgba(0,212,255,0.07)':'rgba(168,85,247,0.07)';
      ctx.fillRect(x1,yTop,xr-x1,h);
      ctx.strokeStyle=isBull?'rgba(0,212,255,0.25)':'rgba(168,85,247,0.25)';
      ctx.lineWidth=1;ctx.setLineDash([2,5]);
      ctx.strokeRect(x1,yTop,xr-x1,h);ctx.setLineDash([]);
    });
    ctx.restore();

    // ── PHASE 2: OB labels in Y-axis gutter (right of plot, no candle overlap) ──
    // Labels are drawn OUTSIDE the plot area in the Y-axis strip.
    // A solid dark background overwrites the plain price tick at that level,
    // replacing it with the OB badge — standard TradingView convention.
    ctx.save();
    ctx.font = 'bold 8px JetBrains Mono';
    ctx.textAlign = 'left';  // drawing from xr leftward into axis

    const LH = 16; // label row height px
    const axisX = xr + 2; // start of Y-axis gutter
    const labels = [];

    // ONE label per OB at zone midpoint, showing type + price range
    obs.forEach(ob => {
      const isBull = ob.type === 'bull';
      const midPrice = (ob.top + ob.bottom) / 2;
      const rawY = y.getPixelForValue(midPrice);
      if (rawY < yt || rawY > yb) return;
      labels.push({
        y: rawY,
        text: `${isBull ? '▲' : '▼'} OB ${ob.bottom.toFixed(0)}–${ob.top.toFixed(0)}`,
        bg: '#0b1117',
        fillColor: isBull ? 'rgba(0,230,118,0.20)' : 'rgba(255,61,87,0.20)',
        strokeColor: isBull ? '#00e676' : '#ff3d57',
        textColor: isBull ? '#00e676' : '#ff3d57',
      });
    });

    // Prev close label at exact price level
    if (prevClose != null) {
      const rawY = y.getPixelForValue(prevClose);
      if (rawY >= yt && rawY <= yb) {
        labels.push({
          y: rawY,
          text: `PC ${prevClose.toFixed(1)}`,
          bg: '#0b1117',
          fillColor: 'rgba(255,196,0,0.15)',
          strokeColor: 'rgba(255,196,0,0.85)',
          textColor: 'rgba(255,196,0,0.95)',
        });
      }
    }

    // Sort top→bottom, then resolve overlaps with two passes
    labels.sort((a, b) => a.y - b.y);
    for (let i = 1; i < labels.length; i++) {
      if (labels[i].y - labels[i-1].y < LH) labels[i].y = labels[i-1].y + LH;
    }
    for (let i = labels.length - 2; i >= 0; i--) {
      if (labels[i+1].y - labels[i].y < LH) labels[i].y = labels[i+1].y - LH;
    }

    // Draw each label in the Y-axis gutter (unclipped so it draws over axis ticks)
    labels.forEach(lb => {
      const tw = ctx.measureText(lb.text).width + 8;
      const lx = axisX;
      const ly = lb.y;
      // Solid dark BG to erase the price tick beneath this label
      ctx.fillStyle = lb.bg;
      ctx.fillRect(lx - 1, ly - 8, tw + 2, 16);
      // Colored badge
      ctx.fillStyle = lb.fillColor;
      ctx.fillRect(lx, ly - 7, tw, 14);
      ctx.strokeStyle = lb.strokeColor;
      ctx.lineWidth = 1;
      ctx.strokeRect(lx + 0.5, ly - 6.5, tw - 1, 13);
      ctx.fillStyle = lb.textColor;
      ctx.fillText(lb.text, lx + 4, ly + 3);
    });

    // CHoCH & BOS — small tags at top strip (above candles, inside chart padding)
    let tagX = xl + 4;
    chochEvts.forEach(e=>{
      const px=tsToX(candles[e.idx]?.timestamp);
      if(px<xl||px>xr) return;
      ctx.fillStyle='rgba(255,145,0,0.18)';
      const tag='CHoCH';
      const tw=ctx.measureText(tag).width+6;
      ctx.fillRect(px-tw/2,yt+2,tw,13);
      ctx.strokeStyle='rgba(255,145,0,0.7)';ctx.lineWidth=1;
      ctx.strokeRect(px-tw/2,yt+2,tw,13);
      ctx.fillStyle='rgba(255,145,0,0.95)';
      ctx.font='bold 8px JetBrains Mono';
      ctx.textAlign='center';
      ctx.fillText(tag,px,yt+12);
    });
    bosEvts.forEach(e=>{
      const px=tsToX(candles[e.idx]?.timestamp);
      if(px<xl||px>xr) return;
      ctx.fillStyle='rgba(192,132,252,0.18)';
      const tag='BOS';
      const tw=ctx.measureText(tag).width+6;
      ctx.fillRect(px-tw/2,yt+17,tw,13);
      ctx.strokeStyle='rgba(192,132,252,0.7)';ctx.lineWidth=1;
      ctx.strokeRect(px-tw/2,yt+17,tw,13);
      ctx.fillStyle='rgba(192,132,252,0.95)';
      ctx.font='bold 8px JetBrains Mono';
      ctx.textAlign='center';
      ctx.fillText(tag,px,yt+27);
    });

    ctx.restore();
  }
};

function updatePriceHeader(candles) {
  if(!candles.length) return;
  const last=candles[candles.length-1],first=candles[0];
  // Calculate change from session open (9:15 IST) not from first candle
  let sessionOpen=first.open;
  // If market is open, use session start
  if(isMarketOpen()) {
    // Session open is always at 9:15
    sessionOpen=first.open;
  }
  const ch=last.close-sessionOpen,pct=(ch/sessionOpen*100);
  document.getElementById('priceDisplay').textContent=last.close.toFixed(2);
  const el=document.getElementById('priceChange');
  el.textContent=`${ch>=0?'+':''}${ch.toFixed(2)} (${pct.toFixed(2)}%)`;
  el.className=`price-change ${ch>=0?'bull':'bear'}`;
  document.getElementById('dayHigh').textContent=Math.max(...candles.map(c=>c.high)).toFixed(2);
  document.getElementById('dayLow').textContent=Math.min(...candles.map(c=>c.low)).toFixed(2);
  const vol=candles.reduce((s,c)=>s+c.volume,0);
  document.getElementById('dayVol').textContent=vol>1e6?(vol/1e6).toFixed(1)+'M':vol>1e3?(vol/1e3).toFixed(0)+'K':vol;
}

function fireAlert(sym, direction) {
  const prev=state.lastSignals[sym];
  if(prev===direction) return;          // no duplicate
  state.lastSignals[sym]=direction;
  if(direction==='WAIT') return;
  if('Notification' in window && Notification.permission==='granted') {
    new Notification(`APEX: ${direction} — ${sym}`, {body:`Signal on ${state.tf}m chart`, icon:''});
  }
  const wrap=document.getElementById('toastWrap');
  const t=document.createElement('div');
  t.className=`toast ${direction==='BUY'?'buy':'sell'}`;
  t.textContent=`${direction} SIGNAL — ${sym}`;
  wrap.appendChild(t);
  setTimeout(()=>t.classList.add('show'),50);
  setTimeout(()=>{t.classList.remove('show');setTimeout(()=>t.remove(),400);},4000);
}

function requestNotificationPermission() {
  if('Notification' in window && Notification.permission==='default') Notification.requestPermission();
}

async function fetchWithTimeout(url, options={}, timeout=7000) {
  const controller = new AbortController();
  const id = setTimeout(() => controller.abort(), timeout);
  try {
    return await fetch(url, {...options, signal: controller.signal});
  } finally {
    clearTimeout(id);
  }
}

// Floor a UTC-ms timestamp to the nearest tf-minute boundary
// e.g. 14:13:00 with tf=5  → 14:10:00
//      14:13:00 with tf=15 → 14:00:00
function floorToInterval(ms, tfMin) {
  const intervalMs = tfMin * 60 * 1000;
  return Math.floor(ms / intervalMs) * intervalMs;
}

async function fetchCandles(symbol, tf) {
  console.log(`[FETCH] fetchCandles(${symbol}, ${tf}m)`);
  const inst = INSTRUMENTS[symbol];
  if (!inst) {
    console.warn(`[FETCH] No instrument for ${symbol}`);
    return null;
  }
  const base = state.proxyUrl.replace(/\/$/, '');
  const oi = OI_SYMBOLS.has(symbol) ? 1 : 0;

  let from, to;
  if (isMarketOpen()) {
    to = new Date();
    from = new Date(to.getTime() - (tf === 5 ? 2 : 5) * 3600000);
    console.log(`[FETCH] Market OPEN, fetching last ${tf === 5 ? 2 : 5} hours from kite/historical`);
  } else {
    const session = getSessionDateRange(symbol, tf);
    from = session.from;
    to = session.to;
    console.log(`[FETCH] Market CLOSED, fetching full session ${from.toISOString()} to ${to.toISOString()} from cached API`);
  }

  const fmt = d => d.toISOString().split('.')[0];
  const dateOnly = d => d.toISOString().split('T')[0];
  const intervalMap = {5:'5minute',15:'15minute',30:'30minute',60:'60minute'};
  const apiInterval = intervalMap[tf] || `${tf}minute`;

  // Minimum candle count varies by interval:
  // 5m session = ~75 candles, 15m session = ~25 candles, 30m = ~13
  const minCandles = Math.max(5, Math.floor(375 / tf) - 5); // 375 min session ÷ tf, minus buffer

  // When market is CLOSED, prefer cached API for complete session data
  if (!isMarketOpen()) {
    try {
      const url = `${base}/api/historical?token=${parseInt(inst.token,10)}&interval=${apiInterval}&from=${dateOnly(from)}&to=${dateOnly(to)}`;
      console.log(`[FETCH] CACHED API GET ${url.substring(0,120)}...`);
      const res = await fetchWithTimeout(url, {credentials:'include'}, 7000);
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const json = await res.json();
      const result = (json.candles || []).map(c => ({
        timestamp: floorToInterval(new Date(c.date || c.timestamp).getTime(), tf),
        open: c.open, high: c.high, low: c.low, close: c.close, volume: c.volume, oi: c.oi || null
      }));
      if (result.length >= minCandles) {
        console.log(`[FETCH] Success: ${result.length} candles from api/historical (${json.source || 'unknown'})`);
        return result;
      }
      console.warn(`[FETCH] Cached API returned only ${result.length} candles (need ${minCandles}), trying kite/historical as fallback...`);
    } catch (e) {
      console.warn(`[FETCH] Cached API failed (${e.message}), trying kite/historical as fallback...`);
    }
  }

  // When market is OPEN or cached API failed, try kite/historical
  try {
    // Use apiInterval string (e.g. '15minute') not raw tf number
    const url = `${base}/kite/historical?instrument_token=${inst.token}&interval=${apiInterval}&from=${fmt(from)}&to=${fmt(to)}&oi=${oi}`;
    console.log(`[FETCH] KITE API GET ${url.substring(0,120)}...`);
    const res = await fetchWithTimeout(url, {credentials:'include'}, 7000);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const json = await res.json();
    if (json.status === 'error') throw new Error(json.message || 'kite/historical error');
    const result = (json.candles || []).map(c => ({
      timestamp: floorToInterval(new Date(c.timestamp).getTime(), tf),
      open: c.open, high: c.high, low: c.low, close: c.close, volume: c.volume, oi: c.oi
    }));
    console.log(`[FETCH] Got ${result.length} candles from kite/historical`);
    if (result.length >= minCandles) {
      console.log(`[FETCH] Success: ${result.length} candles from kite/historical`);
      return result;
    }
    console.warn(`[FETCH] kite/historical returned only ${result.length} candles (need ${minCandles}), trying cached API...`);
  } catch (e) {
    console.warn(`[FETCH] kite/historical failed (${e.message}), trying cached API as fallback...`);
  }

  // Final fallback: cached API
  try {
    const url = `${base}/api/historical?token=${parseInt(inst.token,10)}&interval=${apiInterval}&from=${dateOnly(from)}&to=${dateOnly(to)}`;
    console.log(`[FETCH] FINAL FALLBACK GET ${url.substring(0,120)}...`);
    const res = await fetchWithTimeout(url, {credentials:'include'}, 7000);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const json = await res.json();
    const result = (json.candles || []).map(c => ({
      timestamp: floorToInterval(new Date(c.date || c.timestamp).getTime(), tf),
      open: c.open, high: c.high, low: c.low, close: c.close, volume: c.volume, oi: c.oi || null
    }));
    console.log(`[FETCH] Fallback success: ${result.length} candles from api/historical (${json.source || 'unknown'})`);
    return result;
  } catch (e) {
    console.warn(`[FETCH] All API calls failed (${e.message}), returning null`);
    return null;
  }
}

// ============================================================
async function fetchPrevSessionClose(symbol) {
  try {
    const inst = INSTRUMENTS[symbol];
    if (!inst) return null;
    const base = state.proxyUrl.replace(/\/$/, '');
    // Find the previous trading day relative to the session we're viewing
    const today = getSessionTradingDay();
    const prevDay = new Date(today);
    for (let i = 0; i < 7; i++) {
      prevDay.setDate(prevDay.getDate() - 1);
      const d = prevDay.getDay();
      if (d !== 0 && d !== 6) break;
    }
    const dateOnly = d => d.toISOString().split('T')[0];
    const url = `${base}/api/historical?token=${parseInt(inst.token,10)}&interval=5minute&from=${dateOnly(prevDay)}&to=${dateOnly(prevDay)}`;
    const res = await fetchWithTimeout(url, {credentials:'include'}, 5000);
    if (!res.ok) return null;
    const json = await res.json();
    const cands = json.candles || [];
    if (!cands.length) return null;
    // Last candle close of previous session = prev close
    return cands[cands.length - 1].close ?? null;
  } catch(e) {
    console.warn('[PREV CLOSE] Failed:', e.message);
    return null;
  }
}

async function runPipeline(symbol,tf) {
  console.log(`[PIPELINE] Starting runPipeline for ${symbol} on ${tf}m`);
  document.getElementById('statusText').textContent='LOADING...';
  // Reset holiday offset at start of each fresh pipeline call
  state.sessionDayOffset = 0;
  try {
    // Market hours check
    const banner = document.getElementById('closedBanner');
    if (banner) {
      if(!isMarketOpen()) banner.style.display='block';
      else banner.style.display='none';
    }

    let candles=null, candles15=null;
    if(!state.connected) {
      console.warn(`[PIPELINE] No live Kite session; attempting historical/cached chart data`);
      document.getElementById('statusText').textContent='HISTORICAL MODE · LIVE UNAVAILABLE';
    }

    console.log(`[PIPELINE] Fetching candles...`);
    candles   = await fetchCandles(symbol,tf);
    candles15 = tf===5 ? await fetchCandles(symbol,15) : candles;
    console.log(`[PIPELINE] Got ${candles?.length || 0} candles, ${candles15?.length || 0} candles 15m`);

    // ── HOLIDAY FALLBACK ──────────────────────────────────────────────────
    // If today returned 0 candles (unlisted holiday or pre-market gap),
    // automatically walk back up to 4 previous trading days until we get data.
    // minCandles is tf-aware: 15m session has ~25 candles, not 75 like 5m
    const pipelineMin = Math.max(5, Math.floor(375 / tf) - 5);
    let retryCount = 0;
    while ((!candles || candles.length < pipelineMin) && retryCount < 4) {
      retryCount++;
      state.sessionDayOffset = retryCount;
      const fallbackDay = getSessionTradingDay();
      const fallbackStr = fallbackDay.toLocaleDateString('en-IN',{weekday:'short',day:'numeric',month:'short',year:'numeric',timeZone:'Asia/Kolkata'});
      console.warn(`[PIPELINE] ${candles?.length||0} candles < ${pipelineMin} — possible holiday. Retry ${retryCount}: trying ${fallbackStr}`);
      candles   = await fetchCandles(symbol, tf);
      candles15 = tf===5 ? await fetchCandles(symbol,15) : candles;
      console.log(`[PIPELINE] Retry ${retryCount}: Got ${candles?.length || 0} candles`);
    }
    // ─────────────────────────────────────────────────────────────────────

    if(!candles || candles.length < pipelineMin) {
      console.warn(`[PIPELINE] Insufficient data after retries (${candles?.length || 0} candles, need ${pipelineMin}), aborting render`);
      document.getElementById('statusText').textContent='NO DATA · CHECK CONNECTION';
      state.sessionDayOffset = 0;
      return;
    }

    // Fetch previous session close for chart reference line
    state.prevClose = await fetchPrevSessionClose(symbol);
    console.log(`[PIPELINE] Prev close: ${state.prevClose}`);
    document.getElementById('statusText').textContent = state.connected ? 'LIVE · KITE API' : 'HISTORICAL · CACHED DATA';
    state.candles=candles; state.candles15=candles15;
    console.log(`[PIPELINE] Candles ready: ${candles.length} primary, ${candles15.length} HTF`);

    const closes=candles.map(c=>c.close);
    console.log(`[INDICATORS] Calculating EMA21, EMA50...`);
    const ema21=calcEMA(closes,21),ema50=calcEMA(closes,50);
    const vwap=calcVWAP(candles),rsi=calcRSI(closes,14),macd=calcMACD(closes);
    console.log(`[INDICATORS] EMA21: ${ema21[ema21.length-1]?.toFixed(2)}, EMA50: ${ema50[ema50.length-1]?.toFixed(2)}, RSI: ${rsi[rsi.length-1]?.toFixed(2)}`);
    
    console.log(`[PATTERNS] Detecting order blocks, FVGs...`);
    const obs=detectOrderBlocks(candles),fvgs=detectFVGs(candles);
    const choch=detectCHoCH(candles),bos=detectBOS(candles);
    const chochEvts=detectCHoCHEvents(candles),bosEvts=detectBOSEvents(candles);
    console.log(`[PATTERNS] BOS: ${bos}, CHoCH: ${choch}, OBs: ${obs.length}, FVGs: ${fvgs.length}`);
    
    const volSpike=isVolSpike(candles);
    const htf=htfBias(candles15);
    console.log(`[ANALYSIS] HTF Bias: ${htf}, Vol Spike: ${volSpike}`);
    
    const bias=scoreSignal(candles,ema21,ema50,vwap,rsi,macd,bos,choch,htf,volSpike);
    console.log(`[SIGNAL] Score: ${bias.score}/12, Direction: ${bias.direction} (Bull:${bias.bullScore}, Bear:${bias.bearScore})`);
    
    const tpsl=bias.direction!=='WAIT'?calcTPSL(candles,bias.direction,obs):{entry:closes[closes.length-1].toFixed(2),tp:'—',sl:'—'};
    const lastOI=candles[candles.length-1].oi;

    console.log(`[RENDER] Rendering chart...`);
    renderChart(candles,ema21,ema50,vwap,obs,fvgs,chochEvts,bosEvts);
    console.log(`[RENDER] Updating UI panels...`);
    updatePriceHeader(candles);
    updateIndicatorPanel(ema21,ema50,vwap,rsi,macd,bos,choch,htf,volSpike,bias,lastOI);
    updateSignalCards(bias,tpsl,bos,choch,htf,volSpike);
    updateZoneList(obs,fvgs);
    document.getElementById('symLabel').textContent=INSTRUMENTS[symbol]?.label||symbol;

    // Fire alert if new signal
    fireAlert(symbol,bias.direction);

    // Screener runs in background (don't block pipeline)
    runScreenerBackground();
    console.log(`[PIPELINE] Completed successfully for ${symbol}`);
  } catch (err) {
    console.error(`[PIPELINE] ERROR caught:`, err);
    console.error(`[PIPELINE] Error stack:`, err.stack);
    document.getElementById('statusText').textContent = 'ERROR — LIVE DATA UNAVAILABLE';
    return;
  }
}

async function runScreenerBackground() {
  // This runs in the background without blocking the main pipeline
  try {
    const symbols=Object.keys(INSTRUMENTS).join(',');
    const res=await fetchWithTimeout(`${state.proxyUrl}/kite/screener?interval=${state.tf}&symbols=${symbols}`,{credentials:'include'},7000);
    if(!res.ok) throw new Error(`HTTP ${res.status}`);
    const json=await res.json();
    const results=(json.results||[]).map(r=>({
      sym:r.symbol, price:r.last_price?.toFixed(2)??'—',
      signal:r.signal, reason:r.reason||'—', htf_bias:r.htf_bias||''
    }));
    updateScreener(results);
  } catch(e) {
    // Silent fail - screener is optional
    const fallback=Object.keys(INSTRUMENTS).map(sym=>({sym,price:'—',signal:'WAIT',reason:'Backend unavailable'}));
    updateScreener(fallback);
  }
}

let sseSource=null;

function startTickFeed() {
  if(sseSource) sseSource.close();
  const tokens=Object.values(INSTRUMENTS).map(i=>parseInt(i.token));
  fetch(`${state.proxyUrl}/kite/ws/start`,{
    method:'POST',credentials:'include',
    headers:{'Content-Type':'application/json'},
    body:JSON.stringify({tokens}),
  }).catch(e=>console.warn('ws/start failed:',e.message));

  const base=state.proxyUrl.replace(/\/$/,'');
  sseSource=new EventSource(`${base}/kite/ws/ticks?tokens=${tokens.join(',')}`);
  sseSource.onmessage=(e)=>{
    try {
      const ticks=JSON.parse(e.data);
      const inst=INSTRUMENTS[state.symbol];
      if(inst){
        const tick=ticks[inst.token];
        if(tick&&tick.ltp){
          document.getElementById('priceDisplay').textContent=tick.ltp.toFixed(2);
          const ch=tick.ltp-(tick.ohlc?.close||tick.ltp);
          const pct=tick.ohlc?.close?((ch/tick.ohlc.close)*100).toFixed(2):'0.00';
          const el=document.getElementById('priceChange');
          el.textContent=`${ch>=0?'+':''}${ch.toFixed(2)} (${pct}%)`;
          el.className=`price-change ${ch>=0?'bull':'bear'}`;
          if(tick.oi!=null) {
            document.getElementById('oiDisplay').textContent=Number(tick.oi).toLocaleString('en-IN');
            document.getElementById('oiv').textContent=Number(tick.oi).toLocaleString('en-IN');
          }
          document.getElementById('statusText').textContent='LIVE · TICK FEED';
        }
      }
    } catch(err){console.warn('SSE parse error:',err);}
  };
  sseSource.onerror=()=>{
    document.getElementById('statusText').textContent='LIVE · POLLING';
    sseSource?.close(); sseSource=null;
  };
}

function _renderMacroTile(id, data, decimals) {
  const priceEl = document.getElementById(`macro${id}`);
  const chgEl   = document.getElementById(`macro${id}Chg`);
  if (!priceEl || !chgEl) return;
  if (!data || data.price == null) {
    priceEl.textContent = '—';
    chgEl.textContent   = '—';
    chgEl.className     = 'macro-chg neutral';
    return;
  }
  priceEl.textContent = Number(data.price).toFixed(decimals);
  const pct  = data.change_pct ?? 0;
  const sign = pct >= 0 ? '+' : '';
  chgEl.textContent = `${sign}${pct.toFixed(2)}%`;
  chgEl.className   = `macro-chg ${pct > 0.01 ? 'bull' : pct < -0.01 ? 'bear' : 'neutral'}`;
  priceEl.style.color = pct >= 0 ? 'var(--bull)' : 'var(--bear)';
  setTimeout(() => { priceEl.style.color = ''; }, 600);
}

function connectKite() {
  console.log(`[CONNECT] Checking auth status...`);
  const proxy=document.getElementById('proxyUrl').value.trim();
  if(proxy) state.proxyUrl=proxy;
  requestNotificationPermission();
  fetch(`${state.proxyUrl}/kite/auth/status`,{credentials:'include'})
    .then(r=>r.json())
    .then(data=>{
      console.log(`[CONNECT] Auth response:`, data);
      if(data.status==='ok'){
        document.getElementById('statusText').textContent=`CONNECTED · ${data.user||''}`;
        state.connected=true;
        console.log(`[CONNECT] Connected as ${data.user}`);
        runPipeline(state.symbol,state.tf);
        startTickFeed();
        if(state.refreshTimer) clearInterval(state.refreshTimer);
        state.refreshTimer=setInterval(()=>runPipeline(state.symbol,state.tf),15000);
        // Start macro ticker (5s refresh — authentic Kite feed)
        fetchGlobalQuotes();
        if(macroTimer) clearInterval(macroTimer);
        macroTimer=setInterval(fetchGlobalQuotes, 5000);
      } else {
        console.log(`[CONNECT] Auth failed: ${data.status}, using demo mode`);
        document.getElementById('statusText').textContent='AUTH ERROR — check backend';
        runPipeline(state.symbol,state.tf);
      }
    })
    .catch((err)=>{
      console.log(`[CONNECT] Backend unavailable (${err.message}), using demo mode`);
      document.getElementById('statusText').textContent='DEMO MODE · BACKEND UNREACHABLE';
      runPipeline(state.symbol,state.tf);
    });
}

    _rcStartMin = chart.scales.x.min;
    _rcStartMax = chart.scales.x.max;
  }
});
document.addEventListener('mousemove', e => {
  if (!_rcDrag || !chart) return;
  const rect = _cvs.getBoundingClientRect();
  const plotW = chart.scales.x.right - chart.scales.x.left;
  const range = _rcStartMax - _rcStartMin;
  const dxPx = e.clientX - _rcStartX;
  const dxMs = -(dxPx / plotW) * range;
  let newMin = _rcStartMin + dxMs;
  let newMax = _rcStartMax + dxMs;
  if (SESSION_X_MIN && newMin < SESSION_X_MIN) { newMin = SESSION_X_MIN; newMax = newMin + range; }
  if (SESSION_X_MAX && newMax > SESSION_X_MAX) { newMax = SESSION_X_MAX; newMin = newMax - range; }
  chart.options.scales.x.min = newMin;
  chart.options.scales.x.max = newMax;
  chart.update('none');
});
document.addEventListener('mouseup', e => { if (e.button === 2) _rcDrag = false; });

// Reset zoom button / double-click
document.getElementById('resetZoom').addEventListener('click', () => {
  if (chart) { chart.resetZoom(); if(SESSION_X_MIN){chart.options.scales.x.min=SESSION_X_MIN;chart.options.scales.x.max=SESSION_X_MAX;chart.update('none');} }
});
document.getElementById('mainChart').addEventListener('dblclick', () => {
  if (chart) { chart.resetZoom(); if(SESSION_X_MIN){chart.options.scales.x.min=SESSION_X_MIN;chart.options.scales.x.max=SESSION_X_MAX;chart.update('none');} }
});
