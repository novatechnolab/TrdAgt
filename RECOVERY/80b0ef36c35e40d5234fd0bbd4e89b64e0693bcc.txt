# event_calendar.py — Corporate Event Fetch + Suppression Logic
import logging
import time
from datetime import datetime, timedelta
from config import API_CALL_DELAY_SEC

log = logging.getLogger("event_calendar")

# Cache: {symbol: [event_dict, ...]}
_event_cache = {}
_cache_time = None


def fetch_events(kite) -> dict:
    """
    Fetch corporate events from Kite instruments API.
    Returns dict keyed by symbol with list of upcoming events.
    Call at scan startup and refresh at 12:00 PM.
    """
    global _event_cache, _cache_time
    try:
        events = {}
        # Kite exposes corporate actions per instrument
        # We check next trading day events
        today = datetime.now().date()
        tomorrow = today + timedelta(days=1)
        # Skip weekends for 'tomorrow'
        if tomorrow.weekday() == 5:  # Saturday
            tomorrow = today + timedelta(days=3)
        elif tomorrow.weekday() == 6:  # Sunday
            tomorrow = today + timedelta(days=2)

        # Try fetching from Kite's corporate calendar endpoint
        # Note: This uses the instruments data which includes corporate events
        # Actual implementation depends on broker API version
        try:
            instruments = kite.instruments("NSE")
            time.sleep(API_CALL_DELAY_SEC)
        except Exception as e:
            log.warning(f"Event fetch from Kite failed: {e}")
            _cache_time = datetime.now()
            return {}

        # Kite doesn't expose corporate calendar directly in standard API
        # Flag all symbols with earnings in the lookback using available data
        # In production, integrate with NSE bhavcopy or a data vendor

        _event_cache = events
        _cache_time = datetime.now()
        log.info(f"Event calendar updated: {len(events)} events loaded")
        return events

    except Exception as e:
        log.error(f"Event calendar fetch failed: {e}")
        _cache_time = datetime.now()
        return {}


def get_events_for_symbol(symbol: str) -> list:
    """Return list of event dicts for symbol for next trading day."""
    return _event_cache.get(symbol, [])


def has_event_tomorrow(symbol: str) -> bool:
    return len(get_events_for_symbol(symbol)) > 0


def should_suppress_options_play(symbol: str) -> bool:
    """Return True if options play should be suppressed due to corporate event."""
    return has_event_tomorrow(symbol)


def get_calendar_status() -> dict:
    if _cache_time is None:
        return {"status": "NOT_LOADED", "message": "EVENT DATA UNAVAILABLE"}
    age_min = (datetime.now() - _cache_time).seconds // 60
    return {
        "status": "OK" if age_min < 60 else "STALE",
        "fetched_at": _cache_time.strftime("%H:%M"),
        "age_minutes": age_min,
    }


def is_expiry_day() -> bool:
    """Thursday = expiry day for weekly options."""
    return datetime.now().weekday() == 3  # Thursday = 3

def is_expiry_tomorrow() -> bool:
    """Wednesday = expiry tomorrow warning."""
    return datetime.now().weekday() == 2

def is_last_hour_wednesday() -> bool:
    """Wednesday 14:30–15:30 = expiry day mode."""
    now = datetime.now()
    return now.weekday() == 2 and now.hour >= 14 and now.minute >= 30
